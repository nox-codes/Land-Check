> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the complete LandVerify Express + TypeScript backend with REST API, GraphQL API, auth, AI document analysis, Squad payments, and real-time trust score subscriptions.

**Architecture:** Feature-Based Modules — each domain is a self-contained folder. Shared infrastructure (AI, storage, Squad client, Prisma) lives in dedicated top-level folders. REST at `/api/v1`, GraphQL at `/graphql` with WebSocket subscriptions via `graphql-ws`.

```typescript
import { Request, Response, NextFunction } from 'express';

export interface AppError extends Error {
  statusCode?: number;
}

export function errorHandler(
  err: AppError,
  _req: Request,
  res: Response,
  _next: NextFunction
): void {
  const statusCode = err.statusCode ?? 500;
  res.status(statusCode).json({
    success: false,
    message: err.message ?? 'Internal Server Error',
    error: process.env.NODE_ENV === 'development' ? err.stack : undefined,
  });
}

export function createError(message: string, statusCode: number): AppError {
  const error: AppError = new Error(message);
  error.statusCode = statusCode;
  return error;
}
```

```typescript
import argon2 from 'argon2';
import jwt from 'jsonwebtoken';
import { prisma } from '../../lib/prisma';
import { createError } from '../../middleware/error.middleware';

export interface JwtPayload {
  sub: string;
  role: string;
}

function signToken(userId: string, role: string): string {
  return jwt.sign(
    { sub: userId, role } as JwtPayload,
    process.env.JWT_SECRET!,
    { expiresIn: process.env.JWT_EXPIRES_IN ?? '7d' }
  );
}

export async function registerUser(email: string, password: string) {
  const existing = await prisma.user.findUnique({ where: { email } });
  if (existing) throw createError('Email already registered', 409);

  const passwordHash = await argon2.hash(password);
  const user = await prisma.user.create({ data: { email, passwordHash } });
  const token = signToken(user.id, user.role);
  return { token, user: { id: user.id, email: user.email, role: user.role } };
}

export async function loginUser(email: string, password: string) {
  const user = await prisma.user.findUnique({ where: { email } });
  if (!user || !user.passwordHash) throw createError('Invalid credentials', 401);

  const valid = await argon2.verify(user.passwordHash, password);
  if (!valid) throw createError('Invalid credentials', 401);

  const token = signToken(user.id, user.role);
  return { token, user: { id: user.id, email: user.email, role: user.role } };
}

export function issueTokenForOAuthUser(userId: string, role: string): string {
  return signToken(userId, role);
}
```

```typescript
  try {

  try {
    const { email, password } = req.body;
    if (!email || !password) { res.status(400).json({ success: false, message: 'Email and password are required' }); return; }
    const result = await authService.loginUser(email, password);
    res.status(200).json({ success: true, ...result });

}
```

```typescript
import passport from 'passport';
import { Strategy as LocalStrategy } from 'passport-local';
import { Strategy as JwtStrategy, ExtractJwt } from 'passport-jwt';
import { Strategy as GoogleStrategy } from 'passport-google-oauth20';
import argon2 from 'argon2';
import { prisma } from './prisma';
import type { JwtPayload } from '../modules/auth/auth.service';

passport.use(
  new LocalStrategy({ usernameField: 'email' }, async (email, password, done) => {
    try {
      const user = await prisma.user.findUnique({ where: { email } });
      if (!user || !user.passwordHash) return done(null, false);
      const valid = await argon2.verify(user.passwordHash, password);
      if (!valid) return done(null, false);
      return done(null, user);
    } catch (err) { return done(err); }
  })
);

passport.use(
  new JwtStrategy(
    { jwtFromRequest: ExtractJwt.fromAuthHeaderAsBearerToken(), secretOrKey: process.env.JWT_SECRET! },
    async (payload: JwtPayload, done) => {
      try {
        const user = await prisma.user.findUnique({ where: { id: payload.sub } });
        if (!user) return done(null, false);
        return done(null, user);
      } catch (err) { return done(err); }
    }
  )
);

passport.use(
  new GoogleStrategy(
    {
      clientID: process.env.GOOGLE_CLIENT_ID!,
      clientSecret: process.env.GOOGLE_CLIENT_SECRET!,
      callbackURL: process.env.GOOGLE_CALLBACK_URL!,
    },
    async (_accessToken, _refreshToken, profile, done) => {
      try {
        const email = profile.emails?.[0]?.value;
        if (!email) return done(new Error('No email from Google'), undefined);
        const user = await prisma.user.upsert({
          where: { email },
          update: { googleId: profile.id },
          create: { email, googleId: profile.id },
        });
        return done(null, user);
      } catch (err) { return done(err as Error, undefined); }
    }
  )
);

export default passport;
```

```typescript
import { Request, Response, NextFunction } from 'express';
import passport from '../lib/passport';

export function requireAuth(req: Request, res: Response, next: NextFunction) {
  passport.authenticate('jwt', { session: false }, (err: Error, user: Express.User) => {
    if (err || !user) { res.status(401).json({ success: false, message: 'Unauthorized' }); return; }
    req.user = user;
    next();
  })(req, res, next);
}
```

```typescript
      const fileBuffer = fs.readFileSync(path.resolve(filePath));
      const base64 = fileBuffer.toString('base64');
```

```typescript
export async function uploadDocument(verificationId: string, userId: string, file: MulterFile, documentType: DocumentType) {

  const analysisResult = await ai.analyzeDocument(file.path, documentType);

export async function recomputeTrustScore(verificationId: string, userId: string) {
  const verification = await prisma.landVerification.findFirst({ where: { id: verificationId, userId }, include: { documents: true } });
  if (!verification) throw createError('Verification not found', 404);

  const ai = await getAIProvider();
  const signals = {
    documents: verification.documents.map(d => ({
      authenticityScore: d.authenticityScore ?? 50,
      findings: (d.analysisResult as { findings?: string[] })?.findings ?? [],
      isSuspicious: (d.analysisResult as { isSuspicious?: boolean })?.isSuspicious ?? false,
      rawResponse: '',
    })),
    parcelNumber: verification.parcelNumber,
    ownerName: verification.ownerName,
    location: verification.location,
  };

  const result = await ai.computeTrustScore(verificationId, signals);

  await prisma.trustScoreEvent.create({
    data: { verificationId, previousScore: verification.trustScore, newScore: result.score, reason: result.reason, triggeredBy: 'AI' },
  });

  const updated = await prisma.landVerification.update({ where: { id: verificationId }, data: { trustScore: result.score, status: result.status } });

  pubsub.publish(TRUST_SCORE_UPDATED, {
    trustScoreUpdated: { ...result, id: verificationId, createdAt: new Date().toISOString() },
  });

  return updated;
}
```

```typescript
import { prisma } from '../../lib/prisma';
import * as squad from '../../payments/squad';
import { createError } from '../../middleware/error.middleware';

export async function initiatePayment(params: { verificationId: string; userId: string; amount: number; customerEmail: string; callbackUrl: string }) {
  const verification = await prisma.landVerification.findFirst({ where: { id: params.verificationId, userId: params.userId } });
  if (!verification) throw createError('Verification not found', 404);
  if (!verification.trustScore) throw createError('Verification not yet scored', 400);

  if (verification.status === 'HIGH_RISK') {
    const payment = await prisma.payment.upsert({
      where: { verificationId: params.verificationId },
      update: { status: 'BLOCKED' },
      create: { verificationId: params.verificationId, userId: params.userId, amount: params.amount, status: 'BLOCKED', metadata: { reason: 'HIGH_RISK' } },
    });
    return { status: 'BLOCKED', message: 'Payment blocked due to high fraud risk', payment };
  }

  if (verification.status === 'CAUTION') {
    const user = await prisma.user.findUnique({ where: { id: params.userId } });
    const virtualAccount = await squad.createVirtualAccount({
      verificationId: params.verificationId,
      amount: params.amount,
      customerEmail: params.customerEmail,
      customerName: user?.email ?? params.customerEmail,
    });
    const payment = await prisma.payment.upsert({
      where: { verificationId: params.verificationId },
      update: { squadVirtualAccountId: virtualAccount.virtualAccountNumber, squadReference: virtualAccount.reference, status: 'HELD' },
      create: { verificationId: params.verificationId, userId: params.userId, amount: params.amount, status: 'HELD', squadVirtualAccountId: virtualAccount.virtualAccountNumber, squadReference: virtualAccount.reference, metadata: { virtualAccount } },
    });
    return { status: 'HELD', virtualAccount, payment };
  }

  // VERIFIED — direct charge
  const charge = await squad.initiateCharge({ verificationId: params.verificationId, amount: params.amount, customerEmail: params.customerEmail, callbackUrl: params.callbackUrl });
  const payment = await prisma.payment.upsert({
    where: { verificationId: params.verificationId },
    update: { squadReference: charge.reference, status: 'PENDING' },
    create: { verificationId: params.verificationId, userId: params.userId, amount: params.amount, status: 'PENDING', squadReference: charge.reference },
  });
  return { status: 'PENDING', checkoutUrl: charge.checkoutUrl, payment };
}

export async function getPaymentStatus(verificationId: string, userId: string) {
  const payment = await prisma.payment.findFirst({ where: { verificationId, userId } });
  if (!payment) throw createError('Payment not found', 404);
  return payment;
}

export async function handleWebhook(payload: string, signature: string) {
  if (!squad.verifyWebhookSignature(payload, signature)) throw createError('Invalid webhook signature', 401);

  const event = JSON.parse(payload) as { event: string; data: { transaction_ref: string } };
  if (event.event === 'charge.success') {
    const payment = await prisma.payment.findFirst({ where: { squadReference: event.data.transaction_ref }, include: { verification: true } });
    if (payment && payment.status === 'HELD' && payment.verification.status === 'VERIFIED') {
      if (payment.squadVirtualAccountId) await squad.releaseEscrow(payment.squadVirtualAccountId);
      await prisma.payment.update({ where: { id: payment.id }, data: { status: 'RELEASED' } });
    }
  }
}
```

```typescript
import { withFilter } from 'graphql-subscriptions';
import { prisma } from '../../lib/prisma';
import { getAIProvider } from '../../ai';
import { pubsub, TRUST_SCORE_UPDATED } from '../../lib/pubsub';
import { createError } from '../../middleware/error.middleware';

interface GQLContext { userId?: string; role?: string; }

export const verificationResolvers = {
  Query: {
    async verification(_: unknown, { id }: { id: string }, ctx: GQLContext) {
      if (!ctx.userId) throw createError('Unauthorized', 401);
      return prisma.landVerification.findFirst({ where: { id, userId: ctx.userId }, include: { documents: true, payment: true, scoreHistory: { orderBy: { createdAt: 'desc' } } } });
    },
    async verifications(_: unknown, __: unknown, ctx: GQLContext) {
      if (!ctx.userId) throw createError('Unauthorized', 401);
      return prisma.landVerification.findMany({ where: { userId: ctx.userId }, include: { documents: true, payment: true, scoreHistory: { orderBy: { createdAt: 'desc' } } }, orderBy: { createdAt: 'desc' } });
    },
    async trustScore(_: unknown, { verificationId }: { verificationId: string }, ctx: GQLContext) {
      if (!ctx.userId) throw createError('Unauthorized', 401);
      const v = await prisma.landVerification.findFirst({ where: { id: verificationId, userId: ctx.userId } });
      if (!v) throw createError('Verification not found', 404);
      return { verificationId, score: v.trustScore ?? 0, status: v.status, reason: 'Current score', triggeredBy: 'SYSTEM' };
    },
  },

  Mutation: {
    async runTrustAnalysis(_: unknown, { verificationId }: { verificationId: string }, ctx: GQLContext) {
      if (!ctx.userId) throw createError('Unauthorized', 401);
      const v = await prisma.landVerification.findFirst({ where: { id: verificationId, userId: ctx.userId }, include: { documents: true } });
      if (!v) throw createError('Verification not found', 404);

      const ai = await getAIProvider();
      const signals = {
        documents: v.documents.map(d => ({
          authenticityScore: d.authenticityScore ?? 50,
          findings: (d.analysisResult as { findings?: string[] })?.findings ?? [],
          isSuspicious: (d.analysisResult as { isSuspicious?: boolean })?.isSuspicious ?? false,
          rawResponse: '',
        })),
        parcelNumber: v.parcelNumber,
        ownerName: v.ownerName,
        location: v.location,
      };

      const result = await ai.computeTrustScore(verificationId, signals);

      await prisma.trustScoreEvent.create({
        data: { verificationId, previousScore: v.trustScore, newScore: result.score, reason: result.reason, triggeredBy: 'AI' },
      });
      await prisma.landVerification.update({ where: { id: verificationId }, data: { trustScore: result.score, status: result.status } });

      pubsub.publish(TRUST_SCORE_UPDATED, {
        trustScoreUpdated: { id: verificationId, verificationId, newScore: result.score, previousScore: v.trustScore, reason: result.reason, triggeredBy: 'AI', createdAt: new Date().toISOString() },
      });

      return result;
    },

    async updateTrustScore(_: unknown, { verificationId, score, reason }: { verificationId: string; score: number; reason: string }, ctx: GQLContext) {
      if (ctx.role !== 'ADMIN') throw createError('Forbidden', 403);
      const v = await prisma.landVerification.findUnique({ where: { id: verificationId } });
      if (!v) throw createError('Verification not found', 404);

      const status = score >= 75 ? 'VERIFIED' : score >= 40 ? 'CAUTION' : 'HIGH_RISK';
      await prisma.trustScoreEvent.create({ data: { verificationId, previousScore: v.trustScore, newScore: score, reason, triggeredBy: 'ADMIN' } });
      await prisma.landVerification.update({ where: { id: verificationId }, data: { trustScore: score, status } });

      pubsub.publish(TRUST_SCORE_UPDATED, {
        trustScoreUpdated: { id: verificationId, verificationId, newScore: score, previousScore: v.trustScore, reason, triggeredBy: 'ADMIN', createdAt: new Date().toISOString() },
      });

      return { verificationId, score, status, reason, triggeredBy: 'ADMIN' };
    },
  },

  Subscription: {
    trustScoreUpdated: {
      subscribe: withFilter(
        () => pubsub.asyncIterator([TRUST_SCORE_UPDATED]),
        (payload: { trustScoreUpdated: { verificationId: string } }, variables: { verificationId: string }) =>
          payload.trustScoreUpdated.verificationId === variables.verificationId
      ),
    },
  },
};
```

```typescript
import { ApolloServer } from '@apollo/server';
import { expressMiddleware } from '@apollo/server/express4';
import { ApolloServerPluginDrainHttpServer } from '@apollo/server/plugin/drainHttpServer';
import { makeExecutableSchema } from '@graphql-tools/schema';
import { useServer } from 'graphql-ws/lib/use/ws';
import { WebSocketServer } from 'ws';
import http from 'http';
import jwt from 'jsonwebtoken';
import { verificationTypeDefs } from './schema/verification.typedefs';
import { dashboardTypeDefs } from './schema/dashboard.typedefs';
import { verificationResolvers } from './resolvers/verification.resolvers';
import { dashboardResolvers } from './resolvers/dashboard.resolvers';

export interface GQLContext { userId?: string; role?: string; }

const schema = makeExecutableSchema({
  typeDefs: [verificationTypeDefs, dashboardTypeDefs],
  resolvers: [verificationResolvers, dashboardResolvers],
});

function getContextFromToken(token?: string): GQLContext {
  if (!token) return {};
  try {
    const payload = jwt.verify(token.replace('Bearer ', ''), process.env.JWT_SECRET!) as { sub: string; role: string };
    return { userId: payload.sub, role: payload.role };
  } catch { return {}; }
}

export async function setupGraphQL(httpServer: http.Server, wss: WebSocketServer) {
  const serverCleanup = useServer(
    { schema, context: (ctx) => getContextFromToken(ctx.connectionParams?.authorization as string | undefined) },
    wss
  );

  const apolloServer = new ApolloServer<GQLContext>({
    schema,
    plugins: [
      ApolloServerPluginDrainHttpServer({ httpServer }),
      { async serverWillStart() { return { async drainServer() { await serverCleanup.dispose(); } }; } },
    ],
  });

  await apolloServer.start();

  return expressMiddleware(apolloServer, {
    context: async ({ req }) => getContextFromToken(req.headers.authorization),
  });
}
```
